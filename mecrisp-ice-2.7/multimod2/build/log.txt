    ientry @ romdir @   \ -- entry# sector16k  ok.
    entry_type dup       \ entry# sector16k -- type.size type.size  ok.
    $F0 and itype !       \ type.size type.size -- type.size  ok.
    $F and isize !         \ type.size --  ok.
;  ok.
  ok.
: isbanked ( -- flag )  ok.
    \ A valid name's entry number is in banks  ok.
    false 8 0 ?do      \ 2.  -- flag  ok.
        i bank@         \ flag -- flag bank-config  ok.
        $03FF and        \ Mask off configuration name entry#  ok.
        ientry @          \ entry#  ok.
        = or               \ flag bank-entry# entry# -- flag  ok.
    loop                    \ flag -- flag  ok.
;  ok.
  ok.
  ok.
: setbanks ( -- )  ok.
    \ Use ibank isize, itype and ientry to set bank registers  ok.
    isize @ ibank @ +    \  -- bank+size  ok.
    ibank @               \ bank+size -- bank+size bank  ok.
    ?do                    \ Loop through each image bank  ok.
        $8000               \ InUse flag  ok.
        itype @ 8 lshift or  \ Image type  ok.
        ientry @ $3FF and or  \ Image entry, E flag clear  ok.
        i bank!                \ Set bank configuration  ok.
    loop  ok.
    ibank @ isize @ + 1-         \ Last bank in chain  ok.
    dup bank@ $4000 or            \ lastbank -- lastbank bankval  ok.
    swap bank!                     \ Set bank 'E' end flag  ok.
;  ok.
  ok.
  ok.

: rom2banks ( -- )  ok.
    \ Copy the flash content of a ROM entry to SPRAM  ok.
    ientry @ romdir @    \ -- entry# sector16k  ok.
    entry_image           \ entry# sector16k -- block#  ok.
    romdir @ image_addr    \ block# -- sector16k  ok.
    ibank @ isize @         \ sector16k -- sector16k bank size  ok.
    case                     \ bank sector16k size -- sector16k bank  ok.
        1 of rom2ram          \ sector16k bank --  ok.
        endof  ok.
        2 of rom32k2ram         \ sector16k bank --  ok.
        endof  ok.
        3 of 2dup rom2ram         \ sector16k bank -- sector16k bank  ok.
            1+ swap 1+ swap        \ sector16k bank -- sector16k+1 bank+1  ok.
            rom32k2ram              \ sector16k+1 bank+1 --  ok.
        endof  ok.
        4 of rom64k2ram               \ sector16k bank --  ok.
        endof  ok.
    endcase  ok.
;  ok.
  ok.
: banks2rom ( sector16k -- )  ok.
    \ Copy content of bank(s) back to flash  ok.
    ibank @ swap isize @    \ sector16k -- bank sector16k size  ok.
    case                     \ bank sector16k size -- bank sector16k  ok.
        1 of ram2rom endof    \ bank sector16k --  ok.
        2 of ram32k2rom endof  \ bank sector16k --  ok.
        3 of 2dup ram2rom       \ bank sector16k -- bank sector16k  ok.
            1+ swap 1+ swap      \ bank sector16k -- bank+1 sector16k+1  ok.
            ram32k2rom            \ bank+1 sector16k+1 --  ok.
        endof  ok.
        4 of ram64k2rom endof       \ bank sector16k --  ok.
    endcase  ok.
;  ok.
  ok.
: mkiram ( name -- )  ok.
    \ Convert a RAM into IRAM using the ibank isize variables  ok.
    \ Note: IRAM denoted by $B3DD $DDDE in first four  ok.
    \ nibbles. $DD $B3 $DE $DD  ok.
    ibank @ isize @ +     \ name -- name bank+size  ok.
    ibank @ ?do            \ name bank+size -- name bank+size bank  ok.
            i zeroram       \ Clear memory  ok.
    loop  ok.
    ibank @ $2000 *           \ name -- name ramaddr  ok.
    $B3DD over sram!           \ name ramaddr data ramaddr -- name ramaddr  ok.
    1+ $DDDE swap sram!         \ name data ramaddr+1 -- name  ok.
    2dup IRAM isize @ or         \ name -- name name type.size  ok.
    romdir @ dir_insert           \ name name type.size sector16k -- name block#
  ok.
    romdir @ image_addr            \ name block# sector16k -- name sector16k  ok
.
    banks2rom                       \ name sector16k -- name  ok.
    romdir @ dir_find ientry !       \ name --  ok.
    setbanks                          \ Update entry# in banks  ok.
;  ok.
  ok.
  ok.
  ok.

: listall ( -- )  ok.
    \ For each bank, do  ok.
    \ Print name, type  ok.
    cr ." Bank InUse End Type  Name" cr  ok.
    8 0 ?do  ok.
        i . ."     "       \ Bank number  ok.
        i bank@  ok.
        dup $8000 and  ok.
        0= if  ok.
            ." 0     "  ok.
        else  ok.
            ." 1     "  ok.
        then  ok.
        dup $4000 and  ok.
        0= if  ok.
            ." 0  "  ok.
        else  ok.
            ." 1  "  ok.
        then  ok.
 dup 6 rshift $F0 and  ok.
 over $8000 and 0= if 1 or then  ok.
            case  ok.
                IRAM of ." IRAM " endof  ok.
                ROM of ." ROM   " endof  ok.
                HARD of ." HARD " endof  ok.
                TAKEOVER of ." TAKE " endof  ok.
                ." ---- "  ok.
     endcase  ok.
        $3FF and  ok.
        dup 0 <> if  ok.
             romdir @ goto_entry  ok.
     4 spiflush printstr idle  ok.

: newram ( size bank -- flag )  ok.
    \ 1. Check that RAM will fit, return false if not  ok.
    \ 2. Insure bank(s) are empty, return false if not  ok.
    \ 3. Setup bank(s) configuration  ok.
    \ 4. Return true  ok.
    ibank ! isize !    \ Save arguments  ok.
    0 ientry !          \ Not a valid entry#, regular RAM  ok.
    IRAM itype !         \ Not correct but won't be backed to flash  ok.
    willfit if            \ 1. Will the new RAM fit memory?  ok.
 isempty if         \ 2. Insure banks are empty  ok.
     setbanks        \ 3. Setup bank(s) configuration  ok.
            isok errno !  ok.
            true  ok.
        else  ok.
            nofree errno !  ok.
            false          \ No, banks not empty, return Fail  ok.
        then  ok.
    else  ok.
        nofit errno !  ok.
        false             \ No, won't fit, return Fail  ok.
    then  ok.
;  ok.
  ok.
: newiram ( name size bank -- flag )  ok.
    \ 1. Insure name does not already exist  ok.
    \ 2. Check that image size will fit, return false if not  ok.
    \ 3. Insure bank(s) are empty, return false if not  ok.
    \ 4. Initialize bank(s) as an IRAM  ok.
    \ 5. Save bank(s) to flash as directory entry 'name'  ok.
    \ 6. Return true  ok.
    ibank ! isize !         \ name size bank -- name  ok.
    2dup romdir @ dir_find   \ name -- name entry#  ok.
    ientry !                  \ name entry# -- name  ok.
    isname not if              \ 1.  name -- name  ok.
        isize @ ibank @         \ name -- name size bank  ok.
        newram if                \ 1~4. name size bank -- name  ok.
     mkiram                \ name --  ok.
     isok errno !  ok.
            true  ok.
        else  ok.
            2drop                    \ name --  ok.
            nouniq errno !  ok.
            false  ok.
        then  ok.
    else  ok.
        2drop                  \ name --  ok.
        noname errno !  ok.
        false  ok.
    then  ok.
;  ok.
  ok.
: save ( -- flag )  ok.
    \ 1. Check that name exists in bank(s), return false if not  ok.
    \ 2. Write bank(s) content back to flash  ok.
    \ 3. Return true  ok.
      ok.
    isbanked if           \ 1.  --  ok.
        settypsz           \ Size value needed when writing back  ok.
        ientry @            \ -- entry#  ok.
        8 0 ?do              \ Find first bank  ok.
            dup               \ entry# -- entry# entry#  ok.
            i bank@ $3FF and   \ entry# entry# -- entry# entry# bank-entry#  ok.

            = if                \ entry# entry# bank-entry -- entry#  ok.
                i ibank !        \ Save first bank#  ok.
                leave             \ entry# -- entry#  ok.
            then  ok.
        loop  ok.
 romdir @ entry_image         \ entry# -- block#  ok.
 romdir @ image_addr           \ block# -- sector16k  ok.
        banks2rom                      \ sector16k --  ok.
        isok errno !  ok.
        true  ok.
    else  ok.
        nobank errno !  ok.
        false  ok.
    then  ok.
;  ok.
  ok.
: savename ( name -- flag )  ok.

    dir_find      \ name sector16k -- entry#  ok.
    ientry !       \ entry# --  ok.
    isname if       \ 1.  ok.
        save         \ 2.  -- flag  ok.
        isok errno !  ok.
    else  ok.
        noname errno !  ok.
        false  ok.
    then  ok.
;  ok.

: unplug ( name -- flag )  ok.
    \ 1. Check that name exists, return false if not  ok.
    \ 2. Check that name exists in bank(s), return false if not  ok.
    \ 3. Check if name is IRAM and save if true  ok.
    \ 4. Clear bank(s) configuration that match name  ok.
    \ 5. Return true  ok.
    romdir @ dir_find       \ 1. name -- entry#  ok.
    ientry ! isname if       \ entry# --  ok.
        isbanked if           \ 2.  ok.
            ientry @ romdir @  \ -- entry# sector16k  ok.
            entry_type          \ entry# sector16k -- type.size  ok.
            $F0 and IRAM = if    \ 3.  ok.
                save              \ --  ok.
            then  ok.
            8 0 ?do  ok.
                ientry @           \ -- entry#  ok.
                i bank@ $03FF and   \ entry# -- entry# bank-entry#  ok.
                = if  ok.
                    0 i bank!         \ 4. Clear configuration  ok.
                then  ok.
            loop  ok.
            isok errno !  ok.
            true  ok.
        else  ok.
            nobank errno !  ok.
            false  ok.
        then  ok.
    else  ok.
        noname errno !  ok.
        false  ok.
    then  ok.
;  ok.
  ok.
: unplugall ( -- )  ok.

    loop  ok.
;  ok.
  ok.
