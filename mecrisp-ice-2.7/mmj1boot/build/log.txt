    dup 13 rshift           \ 16KB = 8K Words  ok.

        1+  ok.

    dup 1024 = if      \ ( name entry#  -- name entry# )  ok.
 drop host2ram   \ ( name 1024 -- name size )  ok.
 dup r> +         \ ( name size -- name size type.size )  ok.
 swap >r           \ ( name size type.size -- name type.size )  ok.
 romdir @           \ ( name type.size -- name type.size sector16k )  ok.
 dir_insert          \ ( name type.size sector16k -- block )  ok.
 romdir @ image_addr  \ ( block sector16k -- sector16k )  ok.
 r>                    \ ( sector16k -- sector16k size )  ok.
 case  ok.
     1 of 0 swap ram2rom endof  ok.
     2 of 0 swap ram32k2rom endof  ok.
     3 of 0 swap ram2rom 1 ram32k2rom endof  ok.
     4 of 0 swap ram64k2rom endof  ok.
            drop ." Unknown image size"  ok.
 endcase  ok.
    else               \ Replace entry image  ok.
        -rot 2drop      \ ( name entry# -- entry# )  ok.
        r> drop          \ Don't need supplied type.size  ok.
        dup romdir @      \ ( entry# -- entry# entry# sector16k )  ok.
        entry_type         \ ( entry# entry# sector16k -- entry# type.size )  ok
.

        romdir @             \ ( size entry# -- size entry# sector16k )  ok.
        entry_image           \ ( size entry# sector16k -- size sector16k )  ok.

        image_addr swap        \ ( size sector16k -- sector16k size )  ok.
        case  ok.
            1 of 0 swap ram2rom endof  ok.
            2 of 0 swap ram32k2rom endof  ok.
            3 of 0 swap ram2rom 1 ram32k2rom endof  ok.
            4 of 0 swap ram64k2rom endof  ok.
            drop ." Unknown image size"  ok.
        endcase  ok.
    then  ok.

k.

: bitstream ( slot -- )  ok.
    \ Write FPGA bitstream to flash. Each image is aligned on  ok.
    \ 0x020000 boundaries. Valid `slot` number is 1, 2, or 3.  ok.
    \ Bitstreams start in sector16k: 8, 16, 24  ok.
  ok.
    dup 1 <        \ ( slot -- slot flag ) Less than 1?  ok.
    over 3 >        \ ( slot flag -- slot flag flag ) Greater than 3?  ok.
    or if            \  ok.
        cr ." Out of range, Use 1, 2, or 3"  ok.
        drop           \  ok.
    else                \  ok.
        bitfence *       \ slot * sector16k  ok.
        8 0 ?do           \  ok.
            i onesram      \ Set all of SPRAM to foxes  ok.
        loop                \  ok.
        host2ram             \ ( sector16k -- sector16k ncount ) Read bitstream 
 ok.
        drop dup              \ ( sector16k ncount -- sector16k sector16k )  ok.

        0 swap ram64k2rom      \ Half of SPRAM to flash  ok.
        4 swap ram32k2rom       \ Other half to flash  ok.
    then  ok.
